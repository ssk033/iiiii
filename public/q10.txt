#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
int main() {
    pid_t pid;
    // Step 1: Fork
    pid = fork();
    if (pid < 0)
        exit(1);
    if (pid > 0)
        exit(0);   // parent exits
    // Step 2: Create new session
    if (setsid() < 0)
        exit(1);
    // Step 3: Change working directory
    chdir("/");
    // Step 4: Reset file permissions
    umask(0);
    // Step 5: Close standard file descriptors
    close(0);
    close(1);
    close(2);
    // (Optional) redirect to /dev/null
    open("/dev/null", O_RDWR);
    dup(0);
    dup(0);
    // Daemon work (example loop)
    while (1) {
        // Your background task here
        sleep(5);
    }
    return 0;
}

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
int main() {
    pid_t pid1, pid2;
    int status;
    // First child
    pid1 = fork();
    if (pid1 == 0) {
        printf("Child 1: PID = %d\n", getpid());
        sleep(2);
        return 10;
    }
    // Second child
    pid2 = fork();
    if (pid2 == 0) {
        printf("Child 2: PID = %d\n", getpid());
        sleep(1);
        return 20;
    }
    // Parent using wait()
    pid_t w = wait(&status);
    printf("Parent: wait() returned PID = %d, exit = %d\n",
           w, WEXITSTATUS(status));
    // Parent using waitpid()
    w = waitpid(pid1, &status, 0);
    printf("Parent: waitpid() for PID %d, exit = %d\n",
           w, WEXITSTATUS(status));
    return 0;
}
