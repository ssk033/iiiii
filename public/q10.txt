#include <stdlib.h>
#include <unistd.h>
#include <sys/stat.h>
#include <fcntl.h>

int main() {
    if (fork() > 0) exit(0);

    setsid();
    chdir("/");
    umask(0);

    close(0);
    close(1);
    close(2);

    open("/dev/null", O_RDWR);
    dup(0);
    dup(0);

    while (1)
        sleep(5);

    return 0;
}

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    int status;
    pid_t p1, p2, w;

    if ((p1 = fork()) == 0) {
        printf("Child 1 PID = %d\n", getpid());
        sleep(2);
        return 10;
    }

    if ((p2 = fork()) == 0) {
        printf("Child 2 PID = %d\n", getpid());
        sleep(1);
        return 20;
    }

    w = wait(&status);
    printf("wait(): PID = %d, Exit = %d\n",
           w, WEXITSTATUS(status));

    w = waitpid(p1, &status, 0);
    printf("waitpid(): PID = %d, Exit = %d\n",
           w, WEXITSTATUS(status));

    return 0;
}
