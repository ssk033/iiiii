#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
int main(void) {
    pid_t pid, pid2;
    if ((pid = fork()) < 0) {
        printf("fork error\n");
    } 
    else if (pid == 0) {   // child process
        printf("Child (PID=%d) exiting\n", getpid());
        exit(0);
    } 
    else {
        printf("Parent (PID=%d) sleeping\n", getpid());
        sleep(5);   // child becomes zombie here
        // Solution: collect child
        if ((pid2 = waitpid(pid, NULL, 0)) != pid)
            printf("waitpid error\n");
        printf("Parent collected child PID = %d\n", pid2);
    }
    return 0;
}

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
int main() {
    int pid = fork();
    if (pid == 0) {
        // Child: user-defined environment
        char *env[] = {"VAR1=Hello", "VAR2=World", NULL};
        execle("./echoall", "echoall", "Child Process", NULL, env);
    } 
    else {
        wait(NULL);
        // Parent: default environment
        execl("./echoall", "echoall", "Parent Process", NULL);
    }
    return 0;
}

#include <stdio.h>
extern char **environ;
int main(int argc, char *argv[]) {
    printf("Arguments:\n");
    for (int i = 0; i < argc; i++) {
        printf("argv[%d] = %s\n", i, argv[i]);
    }
    printf("\nEnvironment Variables:\n");
    for (int i = 0; environ[i] != NULL; i++) {
        printf("%s\n", environ[i]);
    }
    return 0;
}
