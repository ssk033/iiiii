#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
int main() {
    int fd1, fd2;
    char buf[] = "abcdef";
    fd1 = open("t12.txt", O_RDWR | O_CREAT, 0644);
    if (fd1 < 0) {
        printf("Error opening file\n");
        return 1;
    }
    fd2 = dup(fd1);   // duplicate fd1
    printf("fd1 = %d, fd2 = %d\n", fd1, fd2);
    write(fd1, buf, 6);
    return 0;
}

#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
int main() {
    int fd1, fd2;
    char buf[] = "abcdef";
    fd1 = open("t12.txt", O_RDWR | O_CREAT, 0644);
    if (fd1 < 0) {
        printf("Error opening file\n");
        return 1;
    }
    fd2 = 5;   // choose any file descriptor
    if (dup2(fd1, fd2) < 0) {   // correct usage
        printf("dup2 error\n");
        return 1;
    }
    printf("fd1 = %d, fd2 = %d\n", fd1, fd2);
    write(fd2, buf, 6);
    return 0;
}

1.
parent.c
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
int main() {
    int pid = fork();
    if (pid == 0) {
        // child executes another program
        execl("./add", "add", "10", "20", NULL);
    } else {
        // parent waits
        wait(NULL);
        printf("Parent: Child completed\n");
    }
    return 0;
}
