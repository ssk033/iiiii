#include <stdio.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>

int main() {
    printf("Old umask = %03o\n", umask(022));

    int fd = open("file.txt", O_CREAT | O_WRONLY, 0777);
    close(fd);

    chmod("file.txt", 0644);

    printf("Permissions changed to 0644\n");

    return 0;
} return 0;
}
//then cc <9a.c> 
//./a.out
//you should see the old umask, new umask, and the permissions of 
the file changed to 0644.
