#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
int main() {
    int fd = open("file.txt", O_RDWR | O_APPEND);
    char buf[100];
    int n = 5;  

    lseek(fd, 0, SEEK_SET);
    read(fd, buf, n);
   
    dup2(fd, 1);
    
    write(1, buf, n);
    close(fd);
    return 0;
}

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main()
{
    int fd = open("file.txt", O_RDWR);
    struct flock f;

    f.l_type = F_WRLCK;
    f.l_whence = SEEK_END;
    f.l_start = -100;
    f.l_len = 100;

    fcntl(fd, F_GETLK, &f);

    if (f.l_type != F_UNLCK)
    {
        printf("Region is locked by PID: %d\n", f.l_pid);
    }
    else
    {
        printf("Region is not locked. Locking now...\n");

        f.l_type = F_WRLCK;
        fcntl(fd, F_SETLK, &f);

        lseek(fd, -50, SEEK_END);

        char buf[51];
        read(fd, buf, 50);
        buf[50] = '\0';

        printf("Last 50 bytes: %s\n", buf);

        f.l_type = F_UNLCK;
        fcntl(fd, F_SETLK, &f);

        printf("Region unlocked\n");
    }

    close(fd);

    return 0;
}
