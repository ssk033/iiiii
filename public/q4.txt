1.
nano echoall.c

2.
#include <stdio.h>

int main(int argc, char *argv[])
{
    for (int i = 0; i < argc; i++)
    {
        printf("argv[%d] = %s\n", i, argv[i]);
    }

    return 0;
}

3.
Compile echoall.c

cc echoall.c -o echoall

4.
nano interpreter.sh

5.
#!/bin/sh
./echoall ARG_FROM_INTERPRETER $1

6.
Make the shell script executable.

chmod +x interpreter.sh

7.
nano parent.c

8.
#include <stdio.h>
#include <unistd.h>

int main()
{
    if (fork() == 0)
    {
        execl("./interpreter.sh",
              "interpreter.sh",
              "ARG1",
              NULL);
    }

    return 0;
}

9.
Compile parent.c

cc parent.c -o parent

10.
Execute the program.

./parent

Output:

argv[0] = ./echoall
argv[1] = ARG_FROM_INTERPRETER
argv[2] = ARG1

Flow of Execution:

parent
  ↓
interpreter.sh ARG1
  ↓
echoall ARG_FROM_INTERPRETER ARG1
  ↓
Displays all arguments

Result:
The program demonstrates how command-line arguments are passed from a parent process to a shell script and then forwarded to another program using execl().
