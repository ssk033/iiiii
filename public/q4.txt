#include <stdio.h>
#include <unistd.h>
int main() {
    // create hard link
    link("file.txt", "hardlink.txt");
    // create soft link
    symlink("file.txt", "softlink.txt");
    printf("Hard link and Soft link created\n");
    return 0;
}
