PART-A

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main() {
    int fd = open("file.txt", O_RDONLY);
    char ch;
    int i = lseek(fd, 0, SEEK_END);

    while (i--) {
        lseek(fd, i, SEEK_SET);
        read(fd, &ch, 1);
        write(1, &ch, 1);
    }

    close(fd);
    return 0;
}

// Hello, World!

// cc q1a.c
//

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main() {
    int fd = open("file.txt", O_RDONLY);
    char ch;

    if (fork() == 0) {      // Child process
        read(fd, &ch, 1);
        printf("Child read: %c\n", ch);
    } else {                // Parent process
        sleep(1);           // Ensure child runs first
        read(fd, &ch, 1);
        printf("Parent read: %c\n", ch);
    }

    return 0;
}
