QUESTION 3(a)
Delete Empty Files from Current Directory

#include <stdio.h>
#include <dirent.h>
#include <sys/stat.h>

int main() {
    DIR *dir = opendir(".");
    struct dirent *d;
    struct stat s;

    while ((d = readdir(dir))) {
        stat(d->d_name, &s);

        if (s.st_size == 0) {
            remove(d->d_name);
            printf("Deleted: %s\n", d->d_name);
        }
    }

    closedir(dir);
    return 0;
}

// Compile:
// cc q3a.c
//
// Create an empty file:
// touch empty.txt
//
// Run:
// ./a.out
//
// Output:
// Deleted: empty.txt
//
// Logic:
// Read files → Check size → Delete if size is 0

QUESTION 3(b)
Implement ls -li

#include <stdio.h>
#include <dirent.h>
#include <sys/stat.h>

int main() {
    DIR *dir = opendir(".");
    struct dirent *d;
    struct stat s;

    while ((d = readdir(dir))) {
        stat(d->d_name, &s);

        printf("%ld %ld %o %ld %s\n",
               s.st_ino,
               s.st_size,
               s.st_mode,
               s.st_nlink,
               d->d_name);
    }

    closedir(dir);
    return 0;
}

// Compile:
// cc q3b.c
//
// Run:
// ./a.out
//
// Output:
// inode size filename
//
// Logic:
// Read files → Get details → Print
