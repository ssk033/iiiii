#include <setjmp.h>
#include <stdlib.h>
static void f1(int, int, int, int);
static void f2(void);
static jmp_buf jb;
static int gv;   // global variable
int main() {
    int av = 2;              // automatic variable
    register int rv = 3;     // register variable
    volatile int vv = 4;     // volatile variable
    static int sv = 5;       // static variable
    gv = 1;
    if (setjmp(jb) != 0) {
        printf("After longjmp:\n");
        printf("gv=%d av=%d rv=%d vv=%d sv=%d\n",
               gv, av, rv, vv, sv);
        exit(0);
    }
    // modify values
    gv = 95;
    av = 96;
    rv = 97;
    vv = 98;
    sv = 99;
    f1(av, rv, vv, sv);
    return 0;
}
static void f1(int i, int j, int k, int l) {
    printf("In f1():\n");
    printf("gv=%d av=%d rv=%d vv=%d sv=%d\n",
           gv, i, j, k, l);
    f2();
}
static void f2(void) {
    longjmp(jb, 1);
}

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdlib.h>
int main(int argc, char *argv[]) {
    int fd1, fd2;
    char buf[1024];
    int n;
    // check arguments
    if (argc != 3) {
        printf("Usage: %s source destination\n", argv[0]);
        exit(1);
    }
    // open source file
    fd1 = open(argv[1], O_RDONLY);
    if (fd1 < 0) {
        perror("Error opening source file");
        exit(1);
    }
    // create destination file
    fd2 = open(argv[2], O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd2 < 0) {
        perror("Error creating destination file");
        close(fd1);
        exit(1);
    }
    // copy data
    while ((n = read(fd1, buf, sizeof(buf))) > 0) {
        if (write(fd2, buf, n) != n) {
            perror("Write error");
            close(fd1);
            close(fd2);
            exit(1);
        }
    }
    if (n < 0) {
        perror("Read error");
    }
    close(fd1);
    close(fd2);
    printf("File copied successfully\n");
    return 0;
}
