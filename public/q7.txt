#include <stdio.h>
#include <setjmp.h>

jmp_buf jb;
int gv;   

int main() {
    int av = 2;
    register int rv = 3;
    volatile int vv = 4;
    static int sv = 5;

    gv = 1;

    if (setjmp(jb) != 0) {
        printf("After longjmp:\n");
        printf("gv=%d av=%d rv=%d vv=%d sv=%d\n",
               gv, av, rv, vv, sv);
        return 0;
    }

  
    gv = 95;
    av = 96;
    rv = 97;
    vv = 98;
    sv = 99;

    printf("Before longjmp:\n");
    printf("gv=%d av=%d rv=%d vv=%d sv=%d\n",
           gv, av, rv, vv, sv);

    longjmp(jb, 1);

    return 0;
}

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdlib.h>

int main(int argc, char *argv[]) {
    int src, dest, n;
    char buf[1024];

    if (argc != 3) {
        printf("Usage: %s source destination\n", argv[0]);
        return 1;
    }

    src = open(argv[1], O_RDONLY);
    if (src < 0) {
        perror("Source file");
        return 1;
    }

    dest = open(argv[2], O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (dest < 0) {
        perror("Destination file");
        close(src);
        return 1;
    }

    while ((n = read(src, buf, sizeof(buf))) > 0) {
        if (write(dest, buf, n) < 0) {
            perror("Write error");
            break;
        }
    }

    if (n < 0)
        perror("Read error");

    close(src);
    close(dest);

    printf("File copied successfully\n");
    return 0;
}
